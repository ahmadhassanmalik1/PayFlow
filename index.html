Programmable Invoice Ledger Schema.

Integrate PayFlow's structured ledger schema into enterprise ERPs, billing microservices, and audit workflows. 

interface PayFlowInvoiceRecord {
  invoiceNumber: string;        // e.g. "PF-2026-0842"
  issueDate: string;            // ISO "YYYY-MM-DD"
  dueDate: string;              // ISO "YYYY-MM-DD"
  currency: "USD" | "EUR" | "GBP" | "PKR";
  status: "PAID" | "UNPAID" | "PARTIAL" | "OVERDUE";
  vendorEntity: {
    legalName: string;
    taxIdOrNTN: string;         // e.g. "US-EIN-94-3829104"
    billingEmail: string;
  };
  clientEntity: {
    companyName: string;
    taxIdOrNTN: string;         // e.g. "UK-VAT-GB84920147"
    apEmail: string;
  };
  lineItems: Array<{
    id: string;
    description: string;
    quantity: number;
    unitRate: number;
    discountPct: number;
  }>;
  taxSchedule: Array<{
    tierName: string;
    ratePct: number;
    mode: "add" | "withhold";
  }>;
  cryptographicSignature: {
    verificationHash: string;   // Deterministic SHA hash
    auditComplianceScore: number; // 0 - 100%
  };
}
